-- 1. СТВОРЕННЯ БАЗИ ДАНИХ
CREATE DATABASE TravelAgency;
GO
USE TravelAgency;
GO

-- 2. СТВОРЕННЯ ТАБЛИЦЬ

-- Таблиця "Маршрути" (Routes)
CREATE TABLE Routes (
    RouteID INT PRIMARY KEY IDENTITY(1,1),
    Country NVARCHAR(50) NOT NULL,
    Climate NVARCHAR(50),
    Duration INT,            -- Тривалість (днів)
    Hotel NVARCHAR(50),      -- Готель
    Price DECIMAL(10, 2) NOT NULL
);

-- Таблиця "Клієнти" (Clients)
CREATE TABLE Clients (
    ClientID INT PRIMARY KEY IDENTITY(1,1),
    LastName NVARCHAR(50) NOT NULL,
    FirstName NVARCHAR(50) NOT NULL,
    MiddleName NVARCHAR(50),
    Address NVARCHAR(100),
    Phone NVARCHAR(20)
);

-- Таблиця "Путівки" (Vouchers)
CREATE TABLE Vouchers (
    VoucherID INT PRIMARY KEY IDENTITY(1,1),
    RouteID INT NOT NULL,
    ClientID INT NOT NULL,
    DepartureDate DATE NOT NULL,
    Quantity INT DEFAULT 1,           -- Кількість людей
    Discount DECIMAL(10, 2) DEFAULT 0, -- Знижка (сума або %)
    TotalCost DECIMAL(10, 2),         -- Підсумкова вартість
    
    CONSTRAINT FK_Vouchers_Routes FOREIGN KEY (RouteID)
        REFERENCES Routes (RouteID),
    CONSTRAINT FK_Vouchers_Clients FOREIGN KEY (ClientID)
        REFERENCES Clients (ClientID)
);
GO
